function initClsDecomposition(root){ const L = 4, H = 4, P = 5; // layers, heads, patches — schematic /* Widths are tight: 16 head columns, a ⊕ in every gutter and one more per layer for the MLP, inside 1164px. The gutter is wide enough for the feed lines to be seen curving into it, which is the point of putting the ⊕ to the right rather than straight above. */ const GUT = 80; // row labels — "patch 1" needs 63px const CW = 42, CGAP = 16, PITCH = CW + CGAP; const MLPGAP = 20, MLP_W = 46; // the layer's own ⊕, past head 4 const OUT_W = 46; // x_CLS, where the stream comes out const GROUP = (H-1)*PITCH + CW + CGAP + MLPGAP; const Y_LAYER = 0, Y_MLP = 22, MLP_H = 22; const Y_CLS = 58, ADD_H = 24, Y_LINE = Y_CLS + ADD_H/2; const Y_HEAD = 88, Y_ROW0 = 110, RPITCH = 30, CH = 25; const stage = root.querySelector('#cdStage'); const ptsEl = root.querySelector('#cdPoints'); const STEPS = [ ['the residual stream', 'The class token is accumulated: every head, and every MLP, adds something to it. Each ⊕ is one of those additions.'], ['the contributions', 'What a head adds is a sum over tokens — each patch contributes a yi,h,l, and they join the stream together.'], ['one tensor', 'Every contribution is indexed three ways: which patch, which head, which layer. Across and down, that is the whole tensor.'], ['sum over patches', 'Collapse each column and you keep the horizontal axis: one vector per head, per layer — what that head contributed. This is the "what".'], ['sum over heads and layers', 'Collapse each row instead and you keep the vertical axis: one vector per patch — what that image location contributed. This is the "where".'], ['convert to text', 'In a vision + language model like CLIP ViT we can convert these contributions to text.'] ]; // patches 1-3 read as one thing, 4-5 as another — the grouping is the point, // and it is what the next slides do head by head const TEXTS = [ { rows:[0,1,2], t:'“A photo of a pyramid”' }, { rows:[3,4], t:'“A photo of a camel”' } ]; const PANELS = { 3: { sym:'y:,h,l = Σi yi,h,l', keeps:'summed over patches · keeps layer and head', paper:'the paper calls this cl,hhead' }, 4: { sym:'yi,:,: = Σl Σh yi,h,l', keeps:'summed over heads and layers · keeps the patch', paper:'the paper calls this citoken' } }; /* What a box holds depends on what has been summed away. One symbol throughout, with a colon for each index that has been summed over — so the label says which contraction produced it. Leaving y_{i,h,l} on a summed box would be a lie the audience could read straight off the slide. */ const boxLabel = (p, h, l, s) => s <= 2 ? 'y' + (p+1) + ',' + (h+1) + ',' + (l+1) + '' : s === 3 ? 'y:,' + (h+1) + ',' + (l+1) + '' // patches summed away : 'y' + (p+1) + ',:,:'; // heads and layers summed away const colX = (l, h) => GUT + l*GROUP + h*PITCH; // left edge of a head column const addX = (l, h) => colX(l, h) + CW + CGAP/2; // ⊕ sits in the gutter right of it const mlpX = l => colX(l, H-1) + CW + CGAP + MLPGAP/2; const rowY = p => Y_ROW0 + p*RPITCH; const TX = 300, TH = 36; // where the text sits; width is its own let step = 0; const boxes = [], adds = [], mlps = [], labs = {}, feeds = [], texts = [], links = []; const svgNS = 'http://www.w3.org/2000/svg'; const svg = document.createElementNS(svgNS, 'svg'); stage.appendChild(svg); const path = (d, cls) => { const p = document.createElementNS(svgNS, 'path'); p.setAttribute('d', d); p.setAttribute('fill', 'none'); p.setAttribute('stroke', 'var(--role-y)'); p.setAttribute('stroke-width', '1'); p.setAttribute('opacity', '0'); p.setAttribute('stroke-linecap', 'round'); if(cls) p.dataset.k = cls; svg.appendChild(p); return p; }; const line = (x1, y1, x2, y2, stroke, w) => { const l = document.createElementNS(svgNS, 'line'); l.setAttribute('x1', x1); l.setAttribute('y1', y1); l.setAttribute('x2', x2); l.setAttribute('y2', y2); l.setAttribute('stroke', stroke); l.setAttribute('stroke-width', w || 1); l.setAttribute('opacity', '0'); svg.appendChild(l); return l; }; function addSymbol(x, cls){ const a = document.createElement('div'); a.className = 'cd-add' + (cls ? ' ' + cls : ''); a.textContent = '⊕'; a.style.cssText = `left:${x-12}px;top:${Y_CLS}px;width:24px;height:${ADD_H}px`; stage.appendChild(a); adds.push(a); return a; } function build(){ const stream = line(GUT - 14, Y_LINE, mlpX(L-1) + MLP_W/2, Y_LINE, 'var(--role-x)', 1.2); stream.id = 'cdStream'; const lab = (key, cls, txt, x, y, w) => { const d = document.createElement('div'); d.className = 'cd-lab ' + cls; d.innerHTML = txt; d.style.cssText = `left:${x}px;top:${y}px` + (w ? `;width:${w}px` : ''); stage.appendChild(d); labs[key] = d; return d; }; for(let l = 0; l < L; l++){ lab('layer' + l, 'layer', 'layer ' + (l + 1), colX(l, 0), Y_LAYER, (H-1)*PITCH + CW).style.textAlign = 'center'; // the MLP: a block above the stream, adding back into it between layers const m = document.createElement('div'); m.className = 'cd-mlp'; m.dataset.l = l; m.innerHTML = 'MLP' + (l + 1) + ''; m.style.cssText = `left:${mlpX(l) - MLP_W/2}px;top:${Y_MLP}px;width:${MLP_W}px;height:${MLP_H}px`; stage.appendChild(m); mlps.push(m); const drop = line(mlpX(l), Y_MLP + MLP_H, mlpX(l), Y_LINE, 'var(--muted-2)', 1); drop.dataset.mlpdrop = '1'; addSymbol(mlpX(l), 'mlp').dataset.mlp = '1'; for(let h = 0; h < H; h++){ lab('head' + l + h, 'head', 'h' + (h + 1), colX(l, h), Y_HEAD, CW) .style.textAlign = 'center'; const tx = addX(l, h); addSymbol(tx); // the trunk every contribution in this column joins, and one feed per box // up to the centre of the ⊕, not to the bottom of its box: stopping at the // box left a visible gap between the contribution and the addition const trunk = line(tx, rowY(P-1) + CH/2 - 10, tx, Y_LINE, 'var(--role-y)', 1); trunk.dataset.trunk = '1'; feeds.push(trunk); for(let p = 0; p < P; p++){ const by = rowY(p) + CH/2, bx = colX(l, h) + CW + 2; feeds.push(path(`M${bx},${by} L${tx - 7},${by} Q${tx},${by} ${tx},${by - 10}`, 'feed')); const y = document.createElement('div'); y.className = 'cd-y'; y.dataset.l = l; y.dataset.h = h; y.dataset.p = p; y.innerHTML = boxLabel(p, h, l, 0); y.style.cssText = `left:${colX(l,h)}px;top:${rowY(p)}px;width:${CW}px;height:${CH}px`; stage.appendChild(y); boxes.push(y); } } } lab('cls', 'row cls', 'x0CLS', 0, Y_LINE - 11, GUT - 14); lab('out', 'out cls', 'xCLS', mlpX(L-1) + MLP_W/2 + 8, Y_LINE - 11, OUT_W).style.textAlign = 'left'; for(let p = 0; p < P; p++) lab('row' + p, 'row', 'patch ' + (p + 1), 0, rowY(p) + 5, GUT - 14); lab('across', 'across', 'heads and layers →', GUT, rowY(P-1) + CH + 8); svg.insertAdjacentHTML('afterbegin', ''); TEXTS.forEach((g, gi) => { const cy = (rowY(g.rows[0]) + rowY(g.rows[g.rows.length-1]))/2 + CH/2; const d = document.createElement('div'); d.className = 'cd-text'; d.dataset.g = gi; d.textContent = g.t; d.style.cssText = `left:${TX}px;top:${cy - TH/2}px;height:${TH}px`; stage.appendChild(d); texts.push(d); g.rows.forEach(p => { const by = rowY(p) + CH/2, bx = GUT + CW + 3; const a = path(`M${bx},${by} C${bx + 90},${by} ${TX - 90},${cy} ${TX - 10},${cy}`, 'link'); a.setAttribute('marker-end', 'url(#cdA)'); links.push(a); }); }); Object.entries(PANELS).forEach(([i, p]) => { const d = document.createElement('div'); d.className = 'cd-panel'; d.dataset.i = i; d.innerHTML = '
' + p.sym + '
' + '
' + p.keeps + '
' + '
' + p.paper + '
'; d.style.cssText = `left:${GUT + 250}px;top:${rowY(1)}px;width:640px`; stage.appendChild(d); }); } /* A collapse moves every box on an axis to one place; all but one then fade, so the pile reads as a sum rather than as a stack. */ function paint(){ [...root.querySelectorAll('#cdSteps .s')].forEach((e,i) => { e.classList.toggle('on', i <= step); e.classList.toggle('now', i === step); }); svg.querySelector('#cdStream').setAttribute('opacity', .55); adds.forEach(a => a.classList.toggle('on', true)); mlps.forEach(m => m.classList.toggle('on', true)); [...svg.querySelectorAll('[data-mlpdrop]')].forEach(l => l.setAttribute('opacity', .45)); feeds.forEach(f => f.setAttribute('opacity', step >= 1 && step <= 2 ? .4 : 0)); texts.forEach(t => t.classList.toggle('on', step >= 5)); links.forEach(l => l.setAttribute('opacity', step >= 5 ? .75 : 0)); boxes.forEach(b => { const l = +b.dataset.l, h = +b.dataset.h, p = +b.dataset.p; let x, y, show; if(step <= 2){ x = colX(l, h); y = rowY(p); show = step >= 1; } else if(step === 3){ x = colX(l, h); y = rowY(0); show = (p === 0); // columns fold up } else { x = GUT; y = rowY(p); show = (l === 0 && h === 0); // rows fold left, } // and stay for step 6 b.style.left = x + 'px'; b.style.top = y + 'px'; b.style.opacity = show ? 1 : 0; if(show) b.innerHTML = boxLabel(p, h, l, step); }); for(let l = 0; l < L; l++){ labs['layer' + l].classList.add('on'); labs['layer' + l].classList.toggle('hot', step === 3); for(let h = 0; h < H; h++){ labs['head' + l + h].classList.add('on'); labs['head' + l + h].classList.toggle('hot', step === 3); } } labs.cls.classList.add('on'); labs.out.classList.add('on'); // the stream has two ends, and both are named for(let p = 0; p < P; p++){ labs['row' + p].classList.toggle('on', step >= 1); labs['row' + p].classList.toggle('hot', step === 4); } labs.across.classList.toggle('on', step >= 2); labs.across.classList.toggle('hot', step === 3); [...stage.querySelectorAll('.cd-panel')].forEach(p => p.classList.toggle('on', +p.dataset.i === step)); points.forEach((e, i) => e.classList.toggle('on', i === step)); } const points = STEPS.map(st => { const e = document.createElement('div'); e.className = 'cd-point'; e.innerHTML = st[1]; ptsEl.appendChild(e); return e; }); function setStep(k){ step = Math.max(0, Math.min(STEPS.length-1, k)); paint(); } root.querySelector('#cdSteps').innerHTML = STEPS.map((s,i) => '
' + (i+1) + ' · ' + s[0] + '
').join(''); [...root.querySelectorAll('#cdSteps .s')].forEach(e => { e.onclick = () => setStep(+e.dataset.i); }); root.querySelector('#cdReset').onclick = () => setStep(0); build(); paint(); HOST.registerSteps(root, { max: STEPS.length-1, get: () => step, set: setStep }); }
