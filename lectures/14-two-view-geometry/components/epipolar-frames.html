two viewscoplanareach framethe motion
Two cameras see the same world point X. The ray from each centre crosses that camera's image plane at x and x′.
The centres, the point and both of its images lie in one plane — they have to, because the two rays meet at X, and two lines that meet span a plane.
One point, but two coordinate vectors: XO is where it is as far as the left camera is concerned, XO′ as far as the right one is.
__EQ_FRAMEREL__
A single rigid motion carries one frame to the other. R is the rotation between the two cameras and t is the baseline — it points to O′ in O’s coordinate frame.

} (function fit(){ const pts = [OA, OB, XW].concat(frameOf(CA), frameOf(CB)); SC = 1; OX = 0; OY = 0; const xs = pts.map(p => view(p)[0]), ys = pts.map(p => view(p)[1]); SC = Math.min((W - 96)/(Math.max(...xs) - Math.min(...xs)), (H - 74)/(Math.max(...ys) - Math.min(...ys))); }; // Cut a segment into the stretches an image plane hides and the stretches it // does not. Two conditions together mean "hidden": the piece is on the far // side of that plane from the eye, AND its projection lands inside the drawn // card. Screen overlap alone is not occlusion — the stretch from a centre out // to its own image point overlaps its card and is in front of it. // // ALL THE SPLITTING IS DONE IN SCREEN SPACE. The first version found the // outline crossings in 2D and then fed those parameters to a 3D interpolation; // perspective is not linear along a line, so the cut landed about 24px past // the edge you can see and each ray stayed dim well after leaving its card. // The plane crossing is still solved in 3D, then converted to a screen // parameter — and which side of that plane a piece is on follows from whether // it sits before or after that parameter. function occlude(A3, B3, cards){ const As = view(A3), Bs = view(B3); const dx = Bs[0]-As[0], dy = Bs[1]-As[1], L = Math.hypot(dx, dy) || 1; const at = s => [As[0] + dx*s, As[1] + dy*s]; const info = cards.map(c => { const dA = dot(sub(A3, c.p0), c.n), dB = dot(sub(B3, c.p0), c.n); let sp = null; if((dA > 0) !== (dB > 0)){ const p = view(add(A3, mul(sub(B3, A3), dA/(dA - dB)))); sp = ((p[0]-As[0])*dx + (p[1]-As[1])*dy)/(L*L); } return { c: c, sp: sp, sideA: dA > 0, sideB: dB > 0 }; }); const ss = [0, 1]; for(const it of info) if(it.sp !== null && it.sp > 0 && it.sp < 1) ss.push(it.sp); for(const c of cards){ for(let i = 0; i < 4; i++){ const a = c.scr[i], b = c.scr[(i+1) % 4]; const ex = b[0]-a[0], ey = b[1]-a[1]; const den = dx*ey - dy*ex; if(Math.abs(den) < 1e-12) continue; const s = ((a[0]-As[0])*ey - (a[1]-As[1])*ex)/den; const u = ((a[0]-As[0])*dy - (a[1]-As[1])*dx)/den; if(s > 0 && s < 1 && u >= 0 && u <= 1) ss.push(s); } } ss.sort((p, q) => p - q); const out = []; for(let i = 0; i + 1 < ss.length; i++){ if(ss[i+1] - ss[i] < 1e-6) continue; const sm = (ss[i] + ss[i+1])/2, m = at(sm); const hidden = info.some(it => { const side = it.sp === null ? it.sideA : (sm < it.sp ? it.sideA : it.sideB); return side !== it.c.eyeSide && inPoly(m, it.c.scr); }); out.push({ a: at(ss[i]), b: at(ss[i+1]), hidden: hidden }); } return out; } // Sutherland-Hodgman against a convex clip polygon. Needed because the part of // the epipolar plane that is IN FRONT of a card must be repainted over that // card's fill, and only the part actually covered by the card — beyond its // edges the plane was never dimmed, and repainting there would overshoot. function clipPoly(subject, clip){ let area = 0; for(let i = 0; i < clip.length; i++){ const a = clip[i], b = clip[(i+1) % clip.length]; area += a[0]*b[1] - b[0]*a[1]; } const sgn = area >= 0 ? 1 : -1; let out = subject; for(let i = 0; i < clip.length && out.length; i++){ const a = clip[i], b = clip[(i+1) % clip.length]; const side = p2 => sgn*((b[0]-a[0])*(p2[1]-a[1]) - (b[1]-a[1])*(p2[0]-a[0])); const inp = out; out = []; for(let j = 0; j < inp.length; j++){ const P = inp[j], Q = inp[(j+1) % inp.length]; const sp = side(P), sq = side(Q); if(sp >= 0) out.push(P); if((sp >= 0) !== (sq >= 0)){ const t = sp/(sp - sq); out.push([P[0] + (Q[0]-P[0])*t, P[1] + (Q[1]-P[1])*t]); } } } return out; } function drawOccluded(A3, B3, cards, col, w, alpha){ for(const seg of occlude(A3, B3, cards)){ g.save(); g.strokeStyle = col; g.lineWidth = seg.hidden ? 1 : w; g.globalAlpha = seg.hidden ? .24 : alpha; poly([seg.a, seg.b]); g.stroke(); g.restore(); } } function poly(pts, close){ g.beginPath(); pts.forEach((p,i) => i ? g.lineTo(p[0], p[1]) : g.moveTo(p[0], p[1])); if(close) g.closePath(); } function label(t, x, y, col, align, size, italic){ g.font = (italic ?
