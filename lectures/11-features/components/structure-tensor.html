function initStructureTensor(root){ const W = 320, H = 240, SW = 208, SH = 156; const R3 = Math.sqrt(3) / 2; const SAT_GAMMA = 2.5; // see paintM: widens the white at a corner const q = s => root.querySelector(s); const cSrc = q('.st-src'), cGx = q('.st-gx'), cGy = q('.st-gy'); const cMw = q('.st-mwin'), cL1 = q('.st-l1'), cL2 = q('.st-l2'), cKey = q('.st-legend'); const xSrc = cSrc.getContext('2d'), xGx = cGx.getContext('2d'), xGy = cGy.getContext('2d'); const xMw = cMw.getContext('2d'); const xL1 = cL1.getContext('2d'), xL2 = cL2.getContext('2d'); const note = q('.st-note'), chips = [...root.querySelectorAll('.steps .s')]; const cols = [...root.querySelectorAll('.st-col')]; const work = document.createElement('canvas'); work.width = W; work.height = H; const xw = work.getContext('2d', {willReadFrequently:true}); const video = document.createElement('video'); video.autoplay = true; video.playsInline = true; video.muted = true; const st = { step:0, stream:null, frozen:false, still:null, gauss:false, rad:3, gain:10, raf:0 }; const gray = new Float32Array(W*H); const A = new Float32Array(W*H), B = new Float32Array(W*H), C = new Float32Array(W*H); const A2 = new Float32Array(W*H), B2 = new Float32Array(W*H), C2 = new Float32Array(W*H); const T = new Float32Array(W*H); const KX = [-1,0,1,-2,0,2,-1,0,1], KY = [-1,-2,-1,0,0,0,1,2,1]; const clamp8 = v => v < 0 ? 0 : v > 255 ? 255 : v; // ---- the colour map, and the whole reason it works ------------------------- // s is the isotropic part and is rotation invariant; (p,q) is the deviatoric // part and turns twice as fast as the image. Three channels 120 degrees apart // in that plane are grey when the deviatoric part vanishes, and sweep hue as // an edge rotates. M is PSD so |(p,q)| <= s and none of them can go negative. // Hue and saturation come from the SHAPE of M, brightness from its SIZE. // Scaling all three channels by the same factor cannot change hue, so // dividing by the largest of them costs nothing and buys a great deal: // averaging an edge over an n-wide window divides M by about n/2 (a line's // share of a box), which left the averaged panel at 28/255 — black — when // both panels shared one linear scale. Now the panel beside it is the same // colours at a brightness set by how large M is, under a gamma. function paintM(d, i, a, b, c, g){ const s = 0.5*(a + c), p = 0.5*(a - c); if(s <= 1e-9){ d[i]=d[i+1]=d[i+2]=0; d[i+3]=255; return; } // sig = |(p,q)| / s = (l1 - l2) / (l1 + l2): the anisotropy of M, 1 for a // pure edge and 0 for a corner. PSD guarantees it lands in [0,1]. // // Raising it to a power widens the white. Two perpendicular edges have // ANTIPODAL deviatoric vectors — a vertical edge sits at (p,q)=(+s,0) and // a horizontal one at (-s,0) — so blending across a right-angled corner // runs straight through the origin and the hue flips by 180 degrees at the // crossing. Left alone that reads as two colours meeting at a line, with // the corner a thread between them rather than a region. The gamma pulls // mixed pixels toward white without touching a pixel that is genuinely on // an edge, because sig^gamma leaves sig = 1 exactly where it was — the // rank-one panel is unchanged by it, pixel for pixel. // // 2.5 is not arbitrary. Harris's own R = det - k(tr)^2 goes positive once // l1/l2 falls below about 23 for k = 0.04, which is sig = 0.92: by his // criterion almost everything less anisotropic than that is already a // corner. sig^2.5 keeps 0.92 at 0.80 and drops a 4:1 ratio to 0.28, so // what the colour calls an edge and what R calls an edge roughly agree. let sig = Math.hypot(p, b) / s; if(sig > 1) sig = 1; const k = sig > 1e-6 ? Math.pow(sig, SAT_GAMMA) / sig : 0; const p2 = p * k, q2 = b * k; const r = s + p2, gr = s - 0.5*p2 + R3*q2, bl = s - 0.5*p2 - R3*q2; const mx = r > gr ? (r > bl ? r : bl) : (gr > bl ? gr : bl); if(mx <= 1e-9){ d[i]=d[i+1]=d[i+2]=0; d[i+3]=255; return; } const lit = Math.pow(Math.min(1, s*g), 0.6) * 255 / mx; d[i] = clamp8(r * lit); d[i+1] = clamp8(gr * lit); d[i+2] = clamp8(bl * lit); d[i+3] = 255; } // ---- the source ------------------------------------------------------------ // Built to exercise the colour map rather than to look like anything: the // disc's rim runs through every edge orientation, so the hue wheel appears // around it, and the squares put two directions in one window. function pattern(){ xw.fillStyle = '#0d1116'; xw.fillRect(0,0,W,H); xw.fillStyle = '#eceae2'; xw.beginPath(); xw.arc(W*0.30, H*0.50, H*0.33, 0, 7); xw.fill(); xw.fillStyle = '#0d1116'; xw.beginPath(); xw.arc(W*0.30, H*0.50, H*0.20, 0, 7); xw.fill(); xw.fillStyle = '#d8cfa8'; xw.fillRect(W*0.60, H*0.12, W*0.20, H*0.26); xw.save(); xw.translate(W*0.72, H*0.72); xw.rotate(0.5); xw.fillStyle = '#8fb7d8'; xw.fillRect(-W*0.10, -H*0.13, W*0.20, H*0.26); xw.restore(); xw.strokeStyle = '#eceae2'; xw.lineWidth = 2; xw.beginPath(); xw.moveTo(W*0.05,H*0.92); xw.lineTo(W*0.95,H*0.86); xw.stroke(); st.still = xw.getImageData(0,0,W,H); } function grab(){ if(st.frozen && st.still){ xw.putImageData(st.still,0,0); return true; } if(st.stream && video.readyState >= 2 && video.videoWidth){ const vw = video.videoWidth, vh = video.videoHeight; const s = Math.min(vw/W, vh/H), sw = W*s, sh = H*s; xw.drawImage(video, (vw-sw)/2, (vh-sh)/2, sw, sh, 0, 0, W, H); return true; } if(st.still){ xw.putImageData(st.still,0,0); return true; } return false; } // ---- one frame ------------------------------------------------------------- function blur(src, dst, k, r){ // separable, normalised: the averaged panel is then literally an average of // the one beside it, and both can share one gain for(let y=0;y= W ? W-1 : x+t; s += src[y*W+xx]*k[t+r]; } T[y*W+x]=s; } for(let y=0;y= H ? H-1 : y+t; s += T[yy*W+x]*k[t+r]; } dst[y*W+x]=s; } } function kernel(r, gauss){ const k = new Float32Array(2*r+1); let sum = 0; const sig = Math.max(r/2, 0.6); for(let t=-r;t<=r;t++){ const v = gauss ?
