inliers — best so far — δ = 0.16 
best ← nothing
for i = 1 … N:
    sample 2 points at random
    fit the line through them
    count the points within δ* of it
    if count > best:
        best ← this line
optional: refit best to all of its inliers
return best
* δ, the inlier threshold, is set from the noise on a good point — wide enough that a correct one falls inside with high probability, say 0.95, and no wider.
function initRansacLine(root){ // Recovered from the source figure by colour-segmenting its dots: the first // fourteen are its inliers (twelve blue, plus the two it ringed as its // sample), the last six its outliers. Normalised to the unit box. const P = [ [0.0439,0.1737],[0.1448,0.3857],[0.2455,0.2797],[0.3685,0.4044], [0.4279,0.5549],[0.5339,0.6368],[0.5894,0.4698],[0.5970,0.5886], [0.6703,0.8298],[0.7106,0.6606],[0.7712,0.9568],[0.7724,0.7441], [0.8927,0.8511],[0.9109,0.9594], [0.2261,0.8933],[0.3270,0.8089],[0.5694,0.0470],[0.6500,0.1949], [0.7712,0.3006],[0.9330,0.2797] ]; const DELTA = 0.16; // iteration one takes two outliers, iteration two the pair the figure ringed const SAMPLE = [[14, 16], [2, 3]]; const FALLBACK = { '--amber':'#e7a44c', '--signal':'#46d6c0', '--ink':'#edebe4', '--muted':'#7e8996', '--muted-2':'#59626e', '--line-soft':'#222933', '--panel':'#171c24' }; const cv = root.querySelector('.rl-plot'), g = cv.getContext('2d'); const nOut = root.querySelector('.rl-n'), bOut = root.querySelector('.rl-b'); const note = root.querySelector('.rl-note'); const lines = [...root.querySelectorAll('.rl-code .ln')]; const foot = root.querySelector('.rl-foot'); const chips = [...root.querySelectorAll('.steps .s')]; const W = 688, H = 688, PAD = 34; const X = x => PAD + x * (W - 2*PAD); const Y = y => H - PAD - y * (H - 2*PAD); // the one predicate the pseudocode names, and the only place a count is made function inliers(pair){ const a = P[pair[0]], b = P[pair[1]]; const vx = b[0]-a[0], vy = b[1]-a[1], L = Math.hypot(vx, vy); const nx = -vy/L, ny = vx/L; return P.map(p => Math.abs((p[0]-a[0])*nx + (p[1]-a[1])*ny) <= DELTA); } const count = pair => inliers(pair).filter(Boolean).length; // what each step has on the plot, and which pseudocode lines are running const STEP = [ { it:-1, line:0, band:0, run:[], show:-1, refit:0 }, { it:-1, line:0, band:0, run:[0,1], show:1, refit:0 }, { it:0, line:0, band:0, run:[2], show:2, refit:0 }, { it:0, line:1, band:0, run:[3], show:3, refit:0 }, { it:0, line:1, band:1, run:[4], show:4, refit:0 }, { it:0, line:1, band:1, run:[5,6], show:6, refit:0 }, { it:1, line:1, band:0, run:[2,3], show:6, refit:0 }, { it:1, line:1, band:1, run:[4,5,6], show:6, refit:0 }, { it:1, line:1, band:1, run:[7], show:7, refit:1 }, { it:1, line:1, band:1, run:[8], show:8, refit:1 } ]; // Orthogonal least squares through the inliers: the residual delta measures // is perpendicular distance, so that is the residual to minimise. The // principal direction of the covariance is the line. function refit(mask){ let n = 0, mx = 0, my = 0; P.forEach((p, i) => { if(mask[i]){ n++; mx += p[0]; my += p[1]; } }); mx /= n; my /= n; let sxx = 0, syy = 0, sxy = 0; P.forEach((p, i) => { if(!mask[i]) return; const dx = p[0]-mx, dy = p[1]-my; sxx += dx*dx; syy += dy*dy; sxy += dx*dy; }); const th = 0.5 * Math.atan2(2*sxy, sxx - syy); return { c:[mx, my], u:[Math.cos(th), Math.sin(th)] }; } let step = 0; function paint(){ const cs = getComputedStyle(root); const tok = n => (cs.getPropertyValue(n) || '').trim() || FALLBACK[n]; const amber = tok('--amber'), sig = tok('--signal'), soft = tok('--line-soft'); const mut = tok('--muted-2'), panel = tok('--panel'); const S = STEP[step]; g.clearRect(0,0,W,H); g.fillStyle = panel; g.fillRect(0,0,W,H); g.strokeStyle = soft; g.lineWidth = 2; g.strokeRect(X(0), Y(1), X(1)-X(0), Y(0)-Y(1)); const pair = S.it >= 0 ? SAMPLE[S.it] : null; const inl = pair ? inliers(pair) : null; if(pair){ // the band and the line run far past the data on purpose, so they have // to be held inside the frame rather than inside the canvas g.save(); g.beginPath(); g.rect(X(0), Y(1), X(1)-X(0), Y(0)-Y(1)); g.clip(); const a = P[pair[0]], b = P[pair[1]]; const vx = b[0]-a[0], vy = b[1]-a[1], L = Math.hypot(vx, vy); const ux = vx/L, uy = vy/L, nx = -uy, ny = ux, R = 3; // the band first, so the points and the line sit on top of it if(S.band){ g.fillStyle = amber; g.globalAlpha = 0.11; g.beginPath(); g.moveTo(X(a[0]-ux*R + nx*DELTA), Y(a[1]-uy*R + ny*DELTA)); g.lineTo(X(a[0]+ux*R + nx*DELTA), Y(a[1]+uy*R + ny*DELTA)); g.lineTo(X(a[0]+ux*R - nx*DELTA), Y(a[1]+uy*R - ny*DELTA)); g.lineTo(X(a[0]-ux*R - nx*DELTA), Y(a[1]-uy*R - ny*DELTA)); g.closePath(); g.fill(); g.globalAlpha = 1; } if(S.line){ // once the refit is up, the two-point line stays but steps back, so the // shift between "fitted to two" and "fitted to fourteen" is the subject g.strokeStyle = amber; g.lineWidth = S.refit ? 2 : 3; g.globalAlpha = S.refit ? 0.55 : 1; g.beginPath(); g.moveTo(X(a[0]-ux*R), Y(a[1]-uy*R)); g.lineTo(X(a[0]+ux*R), Y(a[1]+uy*R)); g.stroke(); g.globalAlpha = 1; } if(S.refit){ const f = refit(inl); g.strokeStyle = sig; g.lineWidth = 3.5; g.beginPath(); g.moveTo(X(f.c[0]-f.u[0]*R), Y(f.c[1]-f.u[1]*R)); g.lineTo(X(f.c[0]+f.u[0]*R), Y(f.c[1]+f.u[1]*R)); g.stroke(); } g.restore(); } P.forEach((p, i) => { const lit = S.band && inl && inl[i]; g.fillStyle = lit ? sig : mut; g.beginPath(); g.arc(X(p[0]), Y(p[1]), 9, 0, 6.2832); g.fill(); if(pair && (i === pair[0] || i === pair[1])){ g.strokeStyle = amber; g.lineWidth = 3.5; g.beginPath(); g.arc(X(p[0]), Y(p[1]), 14, 0, 6.2832); g.stroke(); } }); } const TEXT = [ 'Twenty points. Some of them are on a line and the rest are not, and nothing says which are which.', 'Two points define a line, so a sample of two is the smallest that generates a model. The loop will draw one, score it, and keep the best.', 'Iteration one: two points, chosen at random. These two happen to be outliers.', 'The line they define. Nothing was fitted to anything — two points, one line.', 'Everything within δ of that line, counted. Six, and you can count them on the picture.', 'Six is better than nothing, so it becomes the best so far. Note what did not happen: the line was not adjusted to fit anybody.', 'Round the loop. The same two lines run again — no new algorithm, a new sample.', 'Fourteen this time, and fourteen beats six. Discard the bad sample, keep this one.', 'Optional, and cheap. The winning line was fitted to two points and no more — now that its inliers are known, fit to all fourteen of them. The sample chose the model; the inliers refine it.', 'After N iterations, return the model with the most support. Outliers never agree with each other, so a sample drawn from them cannot win.' ]; function setStep(n){ step = n < 0 ? 0 : n > 9 ? 9 : n; const S = STEP[step]; lines.forEach((el, i) => { el.classList.toggle('on', i <= S.show); el.classList.toggle('now', S.run.indexOf(i) >= 0); }); // δ is named the moment the counting line appears, and stays named foot.classList.toggle('on', step >= 4); chips.forEach((c, i) => c.classList.toggle('on', i <= step)); note.textContent = TEXT[step]; nOut.textContent = S.band ? String(count(SAMPLE[S.it])) : '—'; bOut.textContent = step >= 7 ? String(count(SAMPLE[1])) : step >= 5 ? String(count(SAMPLE[0])) : '—'; paint(); } setStep(0); HOST.registerSteps(root, { max: 9, get: () => step, set: setStep }); }
